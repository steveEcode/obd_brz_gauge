# Flash Address Map for theme-upgrade branch
# ESP32-S3 16MB Flash Layout

0x0        bootloader/bootloader.bin       (Bootloader)
0x8000     partition_table/partition-table.bin  (Partition Table)
0xF000     ota_data_initial.bin            (OTA Data Initial)
0x20000    obd_brz_gauge.bin               (Application Firmware - OTA_0)
0x320000   [OTA_1 slot - not flashed initially]
0x620000   [theme_0 partition - 4MB - optional, loaded at runtime]
0xA20000   bootmedia.bin                   (Boot Animation - 5.875MB)
